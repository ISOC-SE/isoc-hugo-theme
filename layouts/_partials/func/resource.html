{{- /* Resolve an image/file reference: page resource first, then global assets/. Context: dict "page" PAGE (may be nil) "src" STRING. Returns a resource or "". */ -}} {{- $r := "" -}} {{- $src := .src -}} {{- if and $src (not (urls.Parse $src).IsAbs) -}} {{- with .page -}} {{- with .Resources.Get $src }}{{ $r = . }}{{ end -}} {{- end -}} {{- if not $r -}} {{- with resources.Get (strings.TrimPrefix "/" $src) }}{{ $r = . }}{{ end -}} {{- end -}} {{- end -}} {{- return $r -}}
