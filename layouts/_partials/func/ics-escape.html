{{- /* Escape TEXT values for iCalendar (RFC 5545 §3.3.11). */ -}} {{- $s := replace . "\\" "\\\\" -}} {{- $s = replace $s ";" "\\;" -}} {{- $s = replace $s "," "\\," -}} {{- /* Any line break (CRLF, LF or a lone CR) becomes an escaped newline, so a value can never start a new property line. */ -}} {{- $s = replaceRE `\r\n?|\n` `\n` $s -}} {{- return $s -}}
